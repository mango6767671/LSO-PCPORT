cmake_minimum_required(VERSION 3.20)
project(ONLINE161Client VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ONLINE161Client currently targets Windows.")
endif()
add_library(ONLINE161 SHARED src/online161.cpp)
target_include_directories(ONLINE161 PUBLIC include)
target_compile_features(ONLINE161 PRIVATE cxx_std_17)
target_compile_definitions(ONLINE161 PRIVATE ONLINE161_BUILD UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ONLINE161 PRIVATE winhttp)
set_target_properties(ONLINE161 PROPERTIES OUTPUT_NAME "ONLINE161")
